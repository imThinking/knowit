03 · Merge Algorithm

智能合并算法

KnowIt 的核心是智能合并重复和相似内容，而非简单删除。算法分为三个阶段：检测、匹配、合并。

算法流程概览

┌─────────────────────────────────────────────────────────────┐
│  输入：新内容（title + content_text）                       │
└────────────────────────┬────────────────────────────────────┘
                         ↓
┌─────────────────────────────────────────────────────────────┐
│  阶段 1：快速检测（Fast Detection）                         │
│  • URL 去重（完全相同）                                      │
│  • 标题相似度（Levenshtein 距离 > 0.9）                     │
│  • SimHash 指纹（汉明距离 ≤ 3）                             │
└────────────────────────┬────────────────────────────────────┘
                         ↓
┌─────────────────────────────────────────────────────────────┐
│  阶段 2：相似度匹配（Similarity Matching）                   │
│  • 对候选集进行内容相似度计算                                │
│  • 使用 SimHash 或 Sentence-BERT                             │

阶段 1：快速检测（Fast Detection）

1.1 URL 去重

def detect_duplicate_url(new_item: Item, existing: List[Item]) -> Optional[Item]:
    """检测 URL 完全相同的条目"""
    if not new_item.source_url:
        return None

    for item in existing:
        if item.source_url == new_item.source_url:
            return item
    return None

1.2 标题相似度检测

import Levenshtein

def detect_similar_title(new_item: Item, existing: List[Item]) -> List[Tuple[Item, float]]:
    """检测标题相似度（Levenshtein 距离）"""
    similar = []
    for item in existing:
        distance = Levenshtein.distance(new_item.title, item.title)
        max_len = max(len(new_item.title), len(item.title))
        similarity = 1 - (distance / max_len) if max_len > 0 else 0
        if similarity >= 0.9:  # 阈值可配置
            similar.append((item, similarity))
    return sorted(similar, key=lambda x: x[1], reverse=True)

1.3 SimHash 指纹检测

from simhash import Simhash

def compute_simhash(text: str) -> str:
    """计算 SimHash 指纹（64位）"""
    return Simhash(text).value

def detect_near_duplicate(new_item: Item, existing: List[Item]) -> List[Tuple[Item, int]]:
    """检测近重复（汉明距离 ≤ 3）"""
    new_hash = Simhash(new_item.content_text)
    similar = []

    for item in existing:
        if not item.simhash:
            continue
        item_hash = Simhash(int(item.simhash, 16))
        distance = new_hash.distance(item_hash)
        if distance <= 3:  # 汉明距离阈值
            similar.append((item, distance))

    return sorted(similar, key=lambda x: x[1])

阶段 2：相似度匹配（Similarity Matching）

2.1 内容相似度计算（SimHash）

def calculate_content_similarity(text1: str, text2: str) -> float:
    """计算内容相似度 [0.0, 1.0]"""
    hash1 = Simhash(text1)
    hash2 = Simhash(text2)

    # 汉明距离转相似度
    distance = hash1.distance(hash2)
    max_distance = 64  # SimHash 默认 64 位
    similarity = 1 - (distance / max_distance)

    return similarity

2.2 候选集过滤

def find_similar_candidates(
    new_item: Item,
    existing: List[Item],
    threshold: float = 0.75
) -> List[Tuple[Item, float]]:
    """找到所有相似的条目（相似度 > threshold）"""
    candidates = []

    for item in existing:
        # 多策略计算相似度
        sim = calculate_content_similarity(new_item.content_text, item.content_text)

        if sim >= threshold:
            candidates.append((item, sim))

    # 按相似度排序
    return sorted(candidates, key=lambda x: x[1], reverse=True)

阶段 3：智能合并（Smart Merge）

3.1 要点提取（Sentence Segmentation）

import re

def extract_sentences(text: str) -> List[str]:
    """提取句子（中文 + 英文）"""
    # 移除多余空白
    text = re.sub(r'\s+', ' ', text.strip())

    # 分句（句号、问号、感叹号）
    sentences = re.split(r'[。！？.!?]+', text)

    # 过滤空句子和短句（< 10 字符）
    return [s.strip() for s in sentences if len(s.strip()) >= 10]

3.2 句子去重（Sentence Deduplication）

from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

def deduplicate_sentences(sentences: List[str], threshold: float = 0.85) -> List[str]:
    """去重相似句子"""
    if len(sentences) <= 1:
        return sentences

    # 计算 TF-IDF 向量
    vectorizer = TfidfVectorizer(tokenizer=jieba_tokenizer)
    tfidf_matrix = vectorizer.fit_transform(sentences)

    # 计算余弦相似度矩阵
    similarity_matrix = cosine_similarity(tfidf_matrix)

    # 去重：保留第一个，删除相似的
    unique = []
    removed = set()

    for i in range(len(sentences)):
        if i in removed:
            continue

        unique.append(sentences[i])

        # 标记相似的句子
        for j in range(i + 1, len(sentences)):
            if similarity_matrix[i][j] >= threshold:
                removed.add(j)

    return unique

def jieba_tokenizer(text):
    """jieba 分词器"""
    import jieba
    return jieba.lcut(text)

3.3 内容聚类（Content Clustering）

from sklearn.cluster import KMeans

def cluster_sentences(sentences: List[str], n_clusters: int = 5) -> Dict[int, List[str]]:
    """将句子按主题聚类"""
    if len(sentences) <= n_clusters:
        n_clusters = len(sentences)

    # TF-IDF 向量化
    vectorizer = TfidfVectorizer(tokenizer=jieba_tokenizer)
    tfidf_matrix = vectorizer.fit_transform(sentences)

    # K-means 聚类
    kmeans = KMeans(n_clusters=n_clusters, random_state=42)
    clusters = kmeans.fit_predict(tfidf_matrix)

    # 整理结果
    result = {i: [] for i in range(n_clusters)}
    for sentence, cluster_id in zip(sentences, clusters):
        result[cluster_id].append(sentence)

    return result

3.4 生成合并内容（Generate Merged Content）

def merge_items(items: List[Item], strategy: str = 'auto') -> str:
    """合并多个条目为一个"""
    # 1. 提取所有句子
    all_sentences = []
    for item in items:
        sentences = extract_sentences(item.content_text)
        all_sentences.extend(sentences)

    # 2. 去重相似句子
    unique_sentences = deduplicate_sentences(all_sentences)

    # 3. 按主题聚类
    clusters = cluster_sentences(unique_sentences, n_clusters=min(5, len(unique_sentences)))

    # 4. 生成合并内容（按聚类排序）
    merged_sections = []
    for cluster_id in sorted(clusters.keys()):
        sentences = clusters[cluster_id]
        if sentences:
            section = '。'.join(sentences) + '。'
            merged_sections.append(section)

    merged_content = '\n\n'.join(merged_sections)

    # 5. 生成元数据
    titles = [item.title for item in items]
    merged_title = f"【合并】{titles[0]}" if len(titles) == 1 else f"【合并】{len(titles)} 篇内容"

    return merged_title, merged_content

用户确认流程

def confirm_merge(items: List[Item]) -> bool:

04 · Similarity

相似度计算

KnowIt 使用多级相似度计算策略，在速度和准确度之间取得平衡。

相似度计算策略

策略 算法 复杂度 适用场景
快速检测 SimHash（汉明距离） O(1) 初步过滤
中度精确 TF-IDF + 余弦相似度 O(n²) 句子去重
高精度 Sentence-BERT Embedding O(n²) 语义相似（可选）

SimHash 实现细节

SimHash 算法原理
  1. 分词（jieba）
  2. 计算每个词的权重（TF-IDF）
  3. 生成每个词的哈希值（64位）
  4. 加权累加（+weight 或 -weight）
  5. 降维（>0 → 1, <0 → 0）
  6. 得到 64 位指纹
from simhash import Simhash

def compute_content_fingerprint(content: str) -> str:
    """计算内容指纹（64位十六进制）"""
    # 预处理：移除 HTML 标签
    from bs4 import BeautifulSoup
    soup = BeautifulSoup(content, 'html.parser')
    text = soup.get_text(separator=' ', strip=True)

    # 计算 SimHash
    hash_value = Simhash(text)
    return f"{hash_value.value:016x}"

def are_near_duplicate(hash1: str, hash2: str, threshold: int = 3) -> bool:
    """判断是否为近重复（汉明距离 ≤ threshold）"""
    h1 = Simhash(int(hash1, 16))
    h2 = Simhash(int(hash2, 16))
    return h1.distance(h2) <= threshold

相似度缓存策略

优化思路：避免每次都重新计算相似度。

• 条目创建时计算 SimHash 并存储
• 相似度计算结果存入 item_similarities 表
• 定期清理过期缓存（> 30 天）
• 条目更新时失效相关缓存 
def get_similar_items(item_id: str, threshold: float = 0.75) -> List[Tuple[str, float]]:
    """从缓存获取相似条目（miss 时重新计算）"""
    # 1. 尝试从缓存读取
    cached = db.execute("""
        SELECT item_id_2, similarity
        FROM item_similarities
        WHERE item_id_1 = ? AND similarity >= ? AND method = 'simhash'
        ORDER BY similarity DESC
        LIMIT 10
    """, (item_id, threshold)).fetchall()

    if cached:
        return [(row[0], row[1]) for row in cached]

    # 2. 缓存 miss，重新计算
    item = db.get_item(item_id)
    all_items = db.get_all_items()

    similar = []
    for other in all_items:
        if other.id == item_id:
            continue

        sim = calculate_content_similarity(item.content_text, other.content_text)
        if sim >= threshold:
            similar.append((other.id, sim))

            # 写入缓存
            db.execute("""
                INSERT OR REPLACE INTO item_similarities
                (item_id_1, item_id_2, similarity, method)
                VALUES (?, ?, ?, 'simhash')
            """, (item_id, other.id, sim))

    return sorted(similar, key=lambda x: x[1], reverse=True)
05 · Search

全文搜索设计

使用 Meilisearch 作为全文搜索引擎，提供毫秒级搜索响应和中文支持。

Meilisearch 集成

from meilisearch import Client

class SearchService:
    def __init__(self, host: str = "http://localhost:7700"):
        self.client = Client(host)
        self.index = self.client.index("items")

    def index_item(self, item: Item):
        """索引单个条目"""
        document = {
            "id": item.id,
            "title": item.title,
            "content_text": item.content_text,
            "author": item.author or "",
            "source_type": item.source_type,
            "collection_id": item.collection_id or "",
            "created_at": item.created_at.isoformat(),
            "word_count": item.word_count or 0,
        }
        self.index.add_documents([document])

    def search(self, query: str, limit: int = 20, filters: dict = None) -> List[dict]:
        """全文搜索"""
        search_params = {
            "limit": limit,
            "attributesToHighlight": ["title", "content_text"],
        }

        if filters:
            search_params["filter"] = self._build_filters(filters)

        results = self.index.search(query, search_params)
        return results["hits"]

    def _build_filters(self, filters: dict) -> str:
        """构建 Meilisearch 过滤表达式"""
        parts = []
        for key, value in filters.items():
            if isinstance(value, list):
                parts.append(f"{key} IN {value}")
            else:
                parts.append(f"{key} = {value}")
        return " AND ".join(parts) if parts else ""

@app.post("/api/search")
async def search_items(query: SearchQuery):
    """搜索 API"""
    service = SearchService()

    # 构建过滤条件
    filters = {}
    if query.status:
        filters["status"] = query.status
    if query.source_type:
        filters["source_type"] = query.source_type
    if query.collection_id:
        filters["collection_id"] = query.collection_id

    # 执行搜索
    results = service.search(
        query=query.q,
        limit=query.limit or 20,
        filters=filters if filters else None
    )

    # 补充相似内容推荐
    for item in results:
        similar = get_similar_items(item["id"], threshold=0.7, limit=3)
        item["similar"] = [
            {"id": s[0], "similarity": s[1]} for s in similar
        ]

    return {"results": results, "total": len(results)}

from clip_to_kami.scripts.convert import generate_pdf, fill_kami_template

def export_collection_pdf(collection_id: str, output_path: str):
    """导出合集为 Kami PDF"""
    # 1. 获取合集条目
    items = db.get_items_in_collection(collection_id)

    # 2. 合并内容
    merged_content = merge_items_content(items)

    # 3. 填充 Kami 模板
    collection = db.get_collection(collection_id)
    html = fill_kami_template(
        body_content=merged_content,
        title=collection.name,
        author="KnowIt",
        date=datetime.now().strftime("%Y-%m-%d"),
        source=f"合集：{collection.name}"
    )

    # 4. 生成 PDF
    generate_pdf(html, output_path)

    return output_path

def merge_items_content(items: List[Item]) -> str:
    """合并多个条目为 HTML"""
    parts = []

    for item in items:
        # 提取正文（移除元数据）
        soup = BeautifulSoup(item.content_html, 'html.parser')

        # 生成章节标题
        section_html = f"""
        

          
{item.title}

          
来源：{item.source_type} · {item.created_at}

          {str(soup)}
        

        """
        parts.append(section_html)

    return '\n\n'.join(parts)

09 · Performance

性能优化策略

在10 万条目规模下保持良好性能。

性能目标

  • 搜索响应：< 100ms（10 万条目）
  • 相似度计算：< 500ms（单条目）
  • 合并操作：< 2s（3-5 篇内容）
  • PDF 导出：< 5s（10 篇内容）

优化策略

1. SimHash 缓存

• 条目创建时计算 SimHash 并存储
• 相似度查询时先查缓存
• 定期清理过期缓存（> 30 天）

2. 相似度计算批处理

• 使用向量化计算（NumPy）
• 多进程并行计算（Process Pool）
• 只计算近 30 天的条目（旧条目合并概率低）

3. Meilisearch 索引优化

• 只索引必要字段（title, content_text）
• 使用 stop words 减少索引大小
• 定期重建索引（每周）

4. 数据库查询优化

• 添加必要的索引（见数据库设计）
• 使用预编译语句（SQLAlchemy）
• 连接池管理（避免频繁连接）

5. 懒加载策略

• 列表页只显示摘要（不加载完整 content）
• 点击详情时再加载完整内容
• 相似推荐异步加载
